🎯 Problem: Operation Counting & Growth Rates

Demonstrate the concrete difference between linear O(n) scan and quadratic O(n²) pair comparisons for input size n.

Key Algorithmic Insight: Big-O measures work scaling as input grows. When n grows 1,000x from 1,000 to 1,000,000, linear work grows 1,000x, while quadratic work grows 1,000,000x.
day01_introduction_to_dsa_and_go_memory_model.go 
package main

import "fmt"

// CountOps demonstrates O(n) vs O(n^2) operation counts
func CountOps(n int) (int, int) {
	linearOps := 0
	for i := 0; i < n; i++ {
		linearOps++ // Executes exactly n times
	}

	quadraticOps := 0
	for i := 0; i < n; i++ {
		for j := 0; j < n; j++ {
			quadraticOps++ // Executes n * n times
		}
	}
	return linearOps, quadraticOps
}

func main() {
	for _, n := range []int{10, 100, 1000} {
		lin, quad := CountOps(n)
		fmt.Printf("n=%-4d  O(n): %-6d  O(n^2): %d\n", n, lin, quad)
	}
}

🎯 Problem: Best Time to Buy and Sell Stock (LeetCode 121)

Given an array prices where prices[i] is the stock price on day i, maximize your profit by choosing one day to buy and a future day to sell.

Key Algorithmic Insight: Track the lowest buying price seen so far (minPrice) and compute profit = price - minPrice at each step in a single O(n) pass.
day02_arrays_and_strings_in_go.go 
package main

import "fmt"

func maxProfit(prices []int) int {
	if len(prices) == 0 {
		return 0
	}
	minPrice := prices[0]
	maxProfit := 0

	for _, p := range prices[1:] {
		// Could we achieve a greater profit selling today?
		if p-minPrice > maxProfit {
			maxProfit = p - minPrice
		}
		// Update the cheapest buying price seen so far
		if p < minPrice {
			minPrice = p
		}
	}
	return maxProfit
}

func main() {
	prices := []int{7, 1, 5, 3, 6, 4}
	fmt.Println("Max Profit:", maxProfit(prices)) // Output: 5
}

🎯 Problem: Reverse Linked List & Detect Cycle (LeetCode 206 & 141)

1. Reverse a singly linked list in-place in O(n) time and O(1) space. 2. Detect if a linked list contains a cycle using two pointers.

Key Algorithmic Insight: For reversal, maintain prev, curr, next pointers and redirect curr.Next = prev. For cycle detection, slow moves 1 step and fast moves 2 steps; they meet if and only if a cycle exists.
day03_linked_lists_with_go_generics.go 
package main

import "fmt"

type ListNode[T any] struct {
	Val  T
	Next *ListNode[T]
}

// ReverseList reverses a linked list in-place
func ReverseList[T any](head *ListNode[T]) *ListNode[T] {
	var prev *ListNode[T]
	curr := head
	for curr != nil {
		nextTemp := curr.Next
		curr.Next = prev
		prev = curr
		curr = nextTemp
	}
	return prev
}

// HasCycle detects loops using Floyd's Tortoise and Hare
func HasCycle[T any](head *ListNode[T]) bool {
	slow, fast := head, head
	for fast != nil && fast.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
		if slow == fast {
			return true
		}
	}
	return false
}

func main() {
	n3 := &ListNode[int]{Val: 3, Next: nil}
	n2 := &ListNode[int]{Val: 2, Next: n3}
	n1 := &ListNode[int]{Val: 1, Next: n2}

	rev := ReverseList(n1)
	for curr := rev; curr != nil; curr = curr.Next {
		fmt.Printf("%d -> ", curr.Val)
	}
	fmt.Println("nil") // 3 -> 2 -> 1 -> nil
}

🎯 Problem: Valid Parentheses (LeetCode 20)

Given a string s containing '(', ')', '{', '}', '[' and ']', determine if the input string is valid by matching corresponding brackets.

Key Algorithmic Insight: Push opening brackets onto a stack. When a closing bracket arrives, verify that the top element matches; if not or stack is empty, return false.
day04_stacks_and_queues.go 
package main

import "fmt"

func isValid(s string) bool {
	stack := make([]rune, 0, len(s))
	pairs := map[rune]rune{')': '(', ']': '[', '}': '{'}

	for _, ch := range s {
		if open, isClose := pairs[ch]; isClose {
			if len(stack) == 0 || stack[len(stack)-1] != open {
				return false
			}
			stack = stack[:len(stack)-1] // Pop
		} else {
			stack = append(stack, ch) // Push
		}
	}
	return len(stack) == 0
}

func main() {
	fmt.Println("()[]{}:", isValid("()[]{}")) // true
	fmt.Println("(]:", isValid("(]"))         // false
}

You are climbing a staircase with n steps. Each time you can climb 1 or 2 steps. In how many distinct ways can you reach the top?

Key Algorithmic Insight: Ways to step n equals ways(n-1) + ways(n-2). Without memoization, recursion is O(2ⁿ); with memoization or iterative DP, it runs in O(n) time and O(1) space.
day05_recursion_and_call_stacks.go 
package main

import "fmt"

// climbStairs with O(1) space Fibonacci optimization
func climbStairs(n int) int {
	if n <= 2 {
		return n
	}
	prev2, prev1 := 1, 2
	for i := 3; i <= n; i++ {
		curr := prev1 + prev2
		prev2 = prev1
		prev1 = curr
	}
	return prev1
}

func main() {
	fmt.Println("Ways for n=5:", climbStairs(5)) // 8
}

🎯 Problem: Validate Binary Search Tree (LeetCode 98)

Given the root of a binary tree, determine if it is a valid binary search tree (BST).

Key Algorithmic Insight: Every node must satisfy min < node.Val < max. When recursing left, update max; when recursing right, update min.
day06_binary_trees_and_bsts.go 
package main

import "fmt"

type TreeNode struct {
	Val   int
	Left  *TreeNode
	Right *TreeNode
}

func isValidBST(root *TreeNode) bool {
	var validate func(node *TreeNode, minVal, maxVal *int) bool
	validate = func(node *TreeNode, minVal, maxVal *int) bool {
		if node == nil {
			return true
		}
		if minVal != nil && node.Val <= *minVal {
			return false
		}
		if maxVal != nil && node.Val >= *maxVal {
			return false
		}
		return validate(node.Left, minVal, &node.Val) &&
			validate(node.Right, &node.Val, maxVal)
	}
	return validate(root, nil, nil)
}

func main() {
	tree := &TreeNode{
		Val:   10,
		Left:  &TreeNode{Val: 5},
		Right: &TreeNode{Val: 15},
	}
	fmt.Println("Is valid BST:", isValidBST(tree)) // true
}

🎯 Problem: Top K Frequent Elements (LeetCode 347)

Given an integer array nums and an integer k, return the k most frequent elements using a min-heap.

Key Algorithmic Insight: Count frequencies with a map, then push into a Min-Heap of size k. If heap size exceeds k, pop the minimum frequency. Leaves top-k elements in O(n log k) time.
day07_heaps_and_priority_queues.go 
package main

import (
	"container/heap"
	"fmt"
)

type Item struct {
	val, count int
}

type MinHeap []Item

func (h MinHeap) Len() int           { return len(h) }
func (h MinHeap) Less(i, j int) bool { return h[i].count < h[j].count }
func (h MinHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *MinHeap) Push(x any)        { *h = append(*h, x.(Item)) }
func (h *MinHeap) Pop() any {
	old := *h
	n := len(old)
	x := old[n-1]
	*h = old[:n-1]
	return x
}

func topKFrequent(nums []int, k int) []int {
	freq := make(map[int]int)
	for _, n := range nums {
		freq[n]++
	}

	h := &MinHeap{}
	heap.Init(h)

	for val, count := range freq {
		heap.Push(h, Item{val, count})
		if h.Len() > k {
			heap.Pop(h)
		}
	}

	result := make([]int, k)
	for i := k - 1; i >= 0; i-- {
		result[i] = heap.Pop(h).(Item).val
	}
	return result
}

func main() {

package main

import "fmt"

func validPath(n int, edges [][]int, source int, destination int) bool {
	if source == destination {
		return true
	}
	adj := make(map[int][]int, n)
	for _, e := range edges {
		adj[e[0]] = append(adj[e[0]], e[1])
		adj[e[1]] = append(adj[e[1]], e[0])
	}

	visited := make([]bool, n)
	visited[source] = true
	queue := []int{source}

	for len(queue) > 0 {
		curr := queue[0]
		queue = queue[1:]

		if curr == destination {
			return true
		}
		for _, nxt := range adj[curr] {
			if !visited[nxt] {
				visited[nxt] = true
				queue = append(queue, nxt)
			}
		}
	}
	return false
}

func numIslands(grid [][]byte) int {

		return 0

				dfs(i, j)
			}
		}
	}

func merge(nums1 []int, m int, nums2 []int, n int) {
	p1 := m - 1
	p2 := n - 1

	for p2 >= 0 {

			p1--
		} else {

			p2--
		}
		p--
	}
}

package main

import "fmt"

func findKthLargest(nums []int, k int) int {
	target := len(nums) - k

	var quickSelect func(l, r int) int
	quickSelect = func(l, r int) int {
		pivot := nums[r]
		p := l
		for i := l; i < r; i++ {
			if nums[i] <= pivot {
				nums[p], nums[i] = nums[i], nums[p]
				p++
			}
		}
		nums[p], nums[r] = nums[r], nums[p]

		if p == target {
			return nums[p]
		}
		if p < target {
			return quickSelect(p+1, r)
		}
		return quickSelect(l, p-1)
	}

	return quickSelect(0, len(nums)-1)
}

func search(nums []int, target int) int {
	low, high := 0, len(nums)-1

	for low <= high {
		mid := low + (high-low)/2
		if nums[mid] == target {
			return mid
		}

		// Check if left half is sorted
		if nums[low] <= nums[mid] {

				high = mid - 1
			} else {
				low = mid + 1
			}

				low = mid + 1
			} else {
				high = mid - 1
			}
		}
	}
	return -1
}

package main

import "fmt"

func coinChange(coins []int, amount int) int {
	dp := make([]int, amount+1)
	// Initialize with impossible high value
	for i := 1; i <= amount; i++ {
		dp[i] = amount + 1
	}
	dp[0] = 0

	for i := 1; i <= amount; i++ {
		for _, c := range coins {
			if i >= c && dp[i-c]+1 < dp[i] {
				dp[i] = dp[i-c] + 1
			}
		}
	}

	if dp[amount] > amount {
		return -1
	}
	return dp[amount]
}

package main

import "fmt"

func subsets(nums []int) [][]int {
	var result [][]int
	var curr []int

	var backtrack func(start int)
	backtrack = func(start int) {
		// Deep copy current buffer
		subset := make([]int, len(curr))
		copy(subset, curr)
		result = append(result, subset)

			curr = append(curr, nums[i]) // Choose
			backtrack(i + 1)             // Explore
			curr = curr[:len(curr)-1]    // Unchoose
		}
	}

	backtrack(0)
	return result
}

func ConstructorNumArray(nums []int) NumArray {
	n := len(nums)
	tree := make([]int, 4*n)
	var build func(node, start, end int)
	build = func(node, start, end int) {
		if start == end {
			tree[node] = nums[start]

		mid := (start + end) / 2

		build(2*node+1, mid+1, end)

		build(1, 0, n-1)

}

day20_union-find_(disjoint_set_union).go 
package main

import "fmt"

type DSU struct {
	parent []int
	rank   []int
}

func NewDSU(size int) *DSU {
	p := make([]int, size)
	r := make([]int, size)
	for i := range p {
		p[i] = i
	}
	return &DSU{parent: p, rank: r}
}

func (d *DSU) Find(x int) int {
	if d.parent[x] != x {
		d.parent[x] = d.Find(d.parent[x]) // Path compression
	}
	return d.parent[x]
}

func (d *DSU) Union(x, y int) bool {
	rootX, rootY := d.Find(x), d.Find(y)
	if rootX == rootY {
		return false // Already connected!
	}
	if d.rank[rootX] < d.rank[rootY] {
		d.parent[rootX] = rootY
	} else if d.rank[rootX] > d.rank[rootY] {
		d.parent[rootY] = rootX
	} else {
		d.parent[rootY] = rootX
		d.rank[rootX]++
	}
	return true
}

func findRedundantConnection(edges [][]int) []int {
	dsu := NewDSU(len(edges) + 1)
	for _, e := range edges {
		if !dsu.Union(e[0], e[1]) {
			return e
		}
	}
	return nil
}

	heap.Init(pq)

	for pq.Len() > 0 {
		curr := heap.Pop(pq).(NodeDist)
		if _, exists := dist[curr.id]; exists {
			continue
		}
		dist[curr.id] = curr.dist

		for _, e := range adj[curr.id] {
			if _, visited := dist[e.to]; !visited {
				heap.Push(pq, NodeDist{e.to, curr.dist + e.weight})
			}
		}
	}

	if len(dist) < n {
		return -1
	}
	maxTime := 0
	for _, d := range dist {
		if d > maxTime {
			maxTime = d
		}
	}
	return maxTime
}

func main() {
	times := [][]int{{2, 1, 1}, {2, 3, 1}, {3, 4, 1}}
	fmt.Println("Delay time:", networkDelayTime(times, 4, 2)) // 2
}

Day 22 Minimum Spanning Trees
Medium O(E log E) Time O(V + E) Space
Phase 6 · Specialized — Core Concept

Connecting all vertices with minimum total edge weight. The Cut Property. Kruskal's algorithm combining edge sorting and Disjoint Set Union (DSU).

Kruskal's Algorithm Flow: 1. Generate all pairwise edges 2. Sort edges by ascending weight 3. Iterate edges: if endpoints belong to different components (DSU), add to MST 4. Stop when V - 1 edges selected

🎯 Problem: Min Cost to Connect All Points (LeetCode 1584)

Given an array points representing coordinates, return the minimum cost to make all points connected using Manhattan distance.

Key Algorithmic Insight: Build complete graph edges with Manhattan weights, sort using `slices.SortFunc`, and apply Kruskal's with DSU until n - 1 edges are accepted.
day22_minimum_spanning_trees.go 
package main

import (
	"fmt"
	"math"
	"slices"
)

type MSTEdge struct {
	u, v, cost int
}

func minCostConnectPoints(points [][]int) int {
	n := len(points)
	var edges []MSTEdge

	for i := 0; i < n; i++ {
		for j := i + 1; j < n; j++ {
			cost := int(math.Abs(float64(points[i][0]-points[j][0]))) +
				int(math.Abs(float64(points[i][1]-points[j][1])))
			edges = append(edges, MSTEdge{i, j, cost})
		}
	}

	slices.SortFunc(edges, func(a, b MSTEdge) int {
		return a.cost - b.cost
	})

	parent := make([]int, n)
	for i := range parent {
		parent[i] = i
	}
	var find func(x int) int
	find = func(x int) int {
		if parent[x] != x {
			parent[x] = find(parent[x])
		}
		return parent[x]
	}

	totalCost, edgesCount := 0, 0
	for _, e := range edges {
		rootU, rootV := find(e.u), find(e.v)
		if rootU != rootV {
			parent[rootU] = rootV
			totalCost += e.cost
			edgesCount++
			if edgesCount == n-1 {
				break
			}
		}
	}
	return totalCost
}

func main() {
	pts := [][]int{{0, 0}, {2, 2}, {3, 10}, {5, 2}, {7, 0}}
	fmt.Println("Min MST cost:", minCostConnectPoints(pts)) // 20
}

🎯 Problem: Course Schedule II (LeetCode 210)

Return the ordering of courses you should take to finish all courses given prerequisite pairs. If impossible, return an empty array.

day23_topological_sort_and_dags.go 
package main

import "fmt"

func findOrder(numCourses int, prerequisites [][]int) []int {
	adj := make(map[int][]int, numCourses)
	inDegree := make([]int, numCourses)

	for _, p := range prerequisites {
		course, pre := p[0], p[1]
		adj[pre] = append(adj[pre], course)
		inDegree[course]++
	}

	queue := make([]int, 0)
	for i := 0; i < numCourses; i++ {
		if inDegree[i] == 0 {
			queue = append(queue, i)
		}
	}

	order := make([]int, 0, numCourses)
	for len(queue) > 0 {
		curr := queue[0]
		queue = queue[1:]
		order = append(order, curr)

		for _, nxt := range adj[curr] {
			inDegree[nxt]--
			if inDegree[nxt] == 0 {
				queue = append(queue, nxt)
			}
		}
	}

	if len(order) == numCourses {
		return order
	}
	return nil // Cycle detected
}

func main() {
	reqs := [][]int{{1, 0}, {2, 0}, {3, 1}, {3, 2}}
	fmt.Println("Course order:", findOrder(4, reqs)) // [0, 1, 2, 3] or [0, 2, 1, 3]
}

🎯 Problem: Longest Substring Without Repeating Characters (LeetCode 3)

Given a string s, find the length of the longest substring without repeating characters.

Key Algorithmic Insight: Keep track of last seen index of each character using a 256-int slice. When duplicate found at right, advance left = max(left, lastSeen[char] + 1).
day24_sliding_window_pattern.go 
package main

import "fmt"

func lengthOfLongestSubstring(s string) int {
	lastSeen := make([]int, 256)
	for i := range lastSeen {
		lastSeen[i] = -1
	}

	maxLen, left := 0, 0
	for right := 0; right < len(s); right++ {
		char := s[right]
		if lastSeen[char] >= left {
			left = lastSeen[char] + 1
		}
		lastSeen[char] = right
		if right-left+1 > maxLen {
			maxLen = right - left + 1
		}
	}
	return maxLen
}

func main() {
	fmt.Println("Longest len:", lengthOfLongestSubstring("abcabcbb")) // 3 ("abc")
}

day25_two_pointers_technique.go 
package main

import "fmt"

func maxArea(height []int) int {
	left, right := 0, len(height)-1
	maxWater := 0

	for left < right {
		h := min(height[left], height[right])
		w := right - left
		if h*w > maxWater {
			maxWater = h * w
		}
		if height[left] < height[right] {
			left++
		} else {
			right--
		}
	}
	return maxWater
}

Day 26 Divide & Conquer with Goroutines
Hard O(n log n) Time O(n) Space
Phase 7 · Techniques — Core Concept

The Divide, Conquer, Combine paradigm. Harnessing Go's concurrency primitives: running independent subproblems in lightweight goroutines and combining results via channels.

Concurrent Divide & Conquer: [MergeSort Slice] / \ [Goroutine 1: Left] [Goroutine 2: Right] │ │ channel <- left channel <- right \ / [Merge Two Halves]

🎯 Problem: Concurrent Parallel Merge Sort in Go

Sort an integer slice concurrently by delegating recursive left and right sub-arrays to separate goroutines bounded by a recursion depth.

Key Algorithmic Insight: Unbounded goroutine creation causes scheduler overhead. Pass a depth budget: when depth <= 0, fall back to sequential sort.
day26_divide_and_conquer_with_goroutines.go 
package main

import "fmt"

func merge(left, right []int) []int {
	res := make([]int, 0, len(left)+len(right))
	i, j := 0, 0
	for i < len(left) && j < len(right) {
		if left[i] <= right[j] {
			res = append(res, left[i])
			i++
		} else {
			res = append(res, right[j])
			j++
		}
	}
	res = append(res, left[i:]...)
	res = append(res, right[j:]...)
	return res
}

func ConcurrentMergeSort(arr []int, depth int) []int {
	if len(arr) <= 1 {
		return arr
	}
	mid := len(arr) / 2

	if depth <= 0 {
		left := ConcurrentMergeSort(arr[:mid], 0)
		right := ConcurrentMergeSort(arr[mid:], 0)
		return merge(left, right)
	}

	leftChan := make(chan []int)
	rightChan := make(chan []int)

	go func() { leftChan <- ConcurrentMergeSort(arr[:mid], depth-1) }()
	go func() { rightChan <- ConcurrentMergeSort(arr[mid:], depth-1) }()

	left := <-leftChan
	right := <-rightChan
	return merge(left, right)
}

func main() {
	data := []int{38, 27, 43, 3, 9, 82, 10}
	sorted := ConcurrentMergeSort(data, 2)
	fmt.Println("Sorted:", sorted) // [3 9 10 27 38 43 82]
}

KMP Match Avoids Re-evaluation: Haystack: [a b a b c a b a b a b ... ] Needle: [a b a b a b] Mismatch at 'c' -> LPS informs us needle prefix 'a b' is already matched! Slide forward without resetting haystack pointer.

🎯 Problem: Find the Index of the First Occurrence in a String (LeetCode 28)

Given two strings needle and haystack, return the index of the first occurrence of needle in haystack, or -1 if needle is not part of haystack.

Key Algorithmic Insight: Compute the LPS (Longest Prefix Suffix) array for needle in O(m). Use it to jump needle index on mismatch, achieving linear O(n + m) runtime.
day27_advanced_strings_(kmp_and_rabin-karp).go 
package main

import "fmt"

func strStr(haystack string, needle string) int {
	if len(needle) == 0 {
		return 0
	}
	// Build LPS array
	lps := make([]int, len(needle))
	j := 0
	for i := 1; i < len(needle); i++ {
		for j > 0 && needle[i] != needle[j] {
			j = lps[j-1]
		}
		if needle[i] == needle[j] {
			j++
			lps[i] = j
		}
	}

	// Search
	j = 0
	for i := 0; i < len(haystack); i++ {
		for j > 0 && haystack[i] != needle[j] {
			j = lps[j-1]
		}
		if haystack[i] == needle[j] {
			j++
		}
		if j == len(needle) {
			return i - len(needle) + 1
		}
	}
	return -1
}

func main() {
	fmt.Println("Found at index:", strStr("sadbutsad", "sad")) // 0

🎯 Problem: Thread-Safe LRU Cache Implementation (LeetCode 146)

Design a data structure that follows the constraints of a Least Recently Used (LRU) cache with O(1) Get and Put operations that is safe for concurrent access.

Key Algorithmic Insight: Map provides O(1) key lookup. Doubly-linked list allows O(1) eviction of oldest element and O(1) promotion to head. sync.RWMutex guards concurrent goroutines.
day28_system_design_and_dsa_in_production.go 
package main

import (
	"container/list"
	"fmt"
	"sync"
)

type CacheItem struct {
	key, value int
}

type LRUCache struct {
	capacity int
	items    map[int]*list.Element
	list     *list.List
	mu       sync.RWMutex
}

func NewLRUCache(capacity int) *LRUCache {
	return &LRUCache{
		capacity: capacity,
		items:    make(map[int]*list.Element),
		list:     list.New(),
	}
}

func (c *LRUCache) Get(key int) int {
	c.mu.Lock()
	defer c.mu.Unlock()

	if elem, ok := c.items[key]; ok {
		c.list.MoveToFront(elem)
		return elem.Value.(CacheItem).value
	}
	return -1
}

func (c *LRUCache) Put(key int, value int) {
	c.mu.Lock()
	defer c.mu.Unlock()

	if elem, ok := c.items[key]; ok {
		elem.Value = CacheItem{key, value}
		c.list.MoveToFront(elem)
		return
	}

	if c.list.Len() >= c.capacity {
		oldest := c.list.Back()
		if oldest != nil {
			delete(c.items, oldest.Value.(CacheItem).key)
			c.list.Remove(oldest)
		}
	}

	c.items[key] = c.list.PushFront(CacheItem{key, value})
}

func main() {
	cache := NewLRUCache(2)
	cache.Put(1, 100)
	cache.Put(2, 200)
	fmt.Println("Get 1:", cache.Get(1)) // 100 (1 is promoted)
	cache.Put(3, 300)                  // Evicts 2
	fmt.Println("Get 2:", cache.Get(2)) // -1 (evicted!)
}

package main

import "fmt"

func twoSum(numbers []int, target int) []int {
	left := 0
	right := len(numbers) - 1

	for left < right {
		sum := numbers[left] + numbers[right]
		if sum == target {
			return []int{left + 1, right + 1} // 1-indexed
		}
		if sum < target {
			left++
		} else {
			right--
		}
	}
	return nil
}
